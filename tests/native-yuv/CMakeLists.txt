cmake_minimum_required(VERSION 3.20)
project(LensioYuvTests LANGUAGES C CXX)
enable_testing()
add_executable(YuvPlaneSmoke ../YuvPlaneSmoke.cpp)
target_compile_features(YuvPlaneSmoke PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(YuvPlaneSmoke PRIVATE /O2 /W4 /WX)
  target_compile_definitions(YuvPlaneSmoke PRIVATE __SSE2__)
else()
  target_compile_options(YuvPlaneSmoke PRIVATE -O3 -Wall -Wextra -Werror)
endif()
add_test(NAME YuvPlaneSmoke COMMAND YuvPlaneSmoke)
if(DEFINED ENV{JAVA_HOME})
  add_library(lensio-yuv SHARED ../../LensioCam/android/app/src/main/jni/yuv_plane.c)
  target_include_directories(lensio-yuv PRIVATE "$ENV{JAVA_HOME}/include")
  if(WIN32)
    target_include_directories(lensio-yuv PRIVATE "$ENV{JAVA_HOME}/include/win32")
  else()
    target_include_directories(lensio-yuv PRIVATE "$ENV{JAVA_HOME}/include/linux")
  endif()
endif()
