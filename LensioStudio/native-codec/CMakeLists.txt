cmake_minimum_required(VERSION 3.20)
project(LensioVideoCodec LANGUAGES CXX)
add_library(lensio-codec64 SHARED AvcDecoder.cpp PixelKernels.cpp)
target_compile_features(lensio-codec64 PRIVATE cxx_std_17)
set_property(TARGET lensio-codec64 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(lensio-codec64 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(lensio-codec64 PRIVATE /O2 /W4 /Brepro)
target_link_options(lensio-codec64 PRIVATE /DYNAMICBASE /NXCOMPAT /INCREMENTAL:NO /Brepro)
target_link_libraries(lensio-codec64 PRIVATE mfplat mf mfuuid wmcodecdspuuid d3d11 dxgi ole32)

add_executable(GenerateAvcFixture ../../tests/GenerateAvcFixture.cpp)
target_compile_definitions(GenerateAvcFixture PRIVATE UNICODE _UNICODE NOMINMAX)
set_property(TARGET GenerateAvcFixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(GenerateAvcFixture PRIVATE mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32)
